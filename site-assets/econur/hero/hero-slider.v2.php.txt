function econur_hero_slides() {
    $slides = get_option('econur_hero_slides');
    if (!is_array($slides) || !$slides) $slides = econur_hero_default_slides();
    $out = array();
    foreach ($slides as $s) {
        if (empty($s['on']) && isset($s['on'])) continue;
        if (!wp_attachment_is_image((int) ($s['desktop'] ?? 0))) continue;
        $out[] = wp_parse_args($s, array('mobile' => 0, 'eyebrow' => '', 'title' => '', 'text' => '', 'cta' => '', 'url' => ''));
    }
    return $out;
}

/* painted width of the banner per breakpoint (it fills a rounded frame; on small screens the frame is narrower than 2.4:1) */
define('ECONUR_HERO_SIZES', '(min-width: 1520px) 1480px, (min-width: 1280px) calc(104vw - 50px), (min-width: 1024px) calc(114vw - 55px), (min-width: 768px) calc(100vw - 32px), calc(150vw - 48px)');
// WordPress re-adds "auto, " to lazy images when it filters content; remove it again for the hero banners only.
add_filter('wp_content_img_tag', function ($img) {
    return str_replace('sizes="auto, ' . ECONUR_HERO_SIZES . '"', 'sizes="' . ECONUR_HERO_SIZES . '"', $img);
});

add_shortcode('econur_hero', function () {
    $slides = econur_hero_slides();
    $n = count($slides);
    if (!$n) return '';
    $arrow = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true" focusable="false"><path d="M5 12h14M13 6l6 6-6 6"/></svg>';
    // mobile frame ratio: the first slide's own mobile image when there is one, otherwise 16:10 cut from the right of the banner
    $m_ratio = '';
    if (!empty($slides[0]['mobile'])) { $mm = wp_get_attachment_metadata((int) $slides[0]['mobile']); if (!empty($mm['width']) && !empty($mm['height'])) $m_ratio = ' style="--ecn-hero-m-ar:' . (int) $mm['width'] . ' / ' . (int) $mm['height'] . '"'; }

    $o  = econur_hero_css();
    $o .= '<section class="ecn-hsl ecn-hero' . ($m_ratio ? ' has-mobile' : '') . '" aria-roledescription="carousel" aria-label="Econur featured products"' . $m_ratio . '>';
    $o .= '<h1 class="ecn-visually-hidden">Econur: 100% natural, handmade skincare in zero-plastic packaging</h1>';
    $o .= '<div class="ecn-hero-track">';
    foreach ($slides as $i => $s) {
        $first = 0 === $i;
        $d = (int) $s['desktop']; $m = (int) $s['mobile'];
        $alt = trim((string) get_post_meta($d, '_wp_attachment_image_alt', true));
        if ('' === $alt) $alt = wp_strip_all_tags(str_replace("\n", ' ', $s['title']));
        $img = wp_get_attachment_image($d, 'full', false, array(
            'alt' => $alt, 'sizes' => ECONUR_HERO_SIZES, 'decoding' => $first ? 'sync' : 'async',
            'loading' => $first ? 'eager' : 'lazy', 'fetchpriority' => $first ? 'high' : 'low', 'class' => 'ecn-hero-img',
        ));
        $img = str_replace('sizes="auto, ', 'sizes="', $img);
        $src = '';
        if ($m && wp_attachment_is_image($m)) {
            $set = wp_get_attachment_image_srcset($m, 'full');
            $src = '<source media="(max-width: 767px)" srcset="' . esc_attr($set ? $set : wp_get_attachment_image_url($m, 'full')) . '" sizes="100vw">';
        }
        $url = $s['url'] ? (0 === strpos($s['url'], '/') ? home_url($s['url']) : $s['url']) : '';
        $o .= '<div class="ecn-hero-slide' . ($first ? ' is-on' : '') . ($src ? ' has-m' : '') . '" role="group" aria-roledescription="slide" aria-label="' . ($i + 1) . ' of ' . $n . '"' . ($first ? '' : ' aria-hidden="true" inert') . '>';
        $o .= '<div class="ecn-hero-media"><picture>' . $src . $img . '</picture></div>';
        $o .= '<div class="ecn-hero-copy">';
        if ($s['eyebrow']) $o .= '<p class="ecn-hero-eyebrow">' . esc_html($s['eyebrow']) . '</p>';
        if ($s['title'])   $o .= '<h2 class="ecn-hero-title">' . nl2br(esc_html($s['title']), false) . '</h2>';
        if ($s['text'])    $o .= '<p class="ecn-hero-text">' . esc_html($s['text']) . '</p>';
        if ($s['cta'] && $url) $o .= '<a class="ecn-hero-cta" href="' . esc_url($url) . '"><span>' . esc_html($s['cta']) . '</span>' . $arrow . '</a>';
        $o .= '</div></div>';
    }
    $o .= '</div>';
    if ($n > 1) {
        $o .= '<button class="ecn-hero-arrow ecn-hero-prev" type="button" aria-label="Previous slide"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true" focusable="false"><path d="m15 6-6 6 6 6"/></svg></button>';
        $o .= '<button class="ecn-hero-arrow ecn-hero-next" type="button" aria-label="Next slide"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true" focusable="false"><path d="m9 6 6 6-6 6"/></svg></button>';
        $o .= '<div class="ecn-hero-ctl"><div class="ecn-hero-dots">';
        foreach ($slides as $i => $s) $o .= '<button type="button" aria-label="Show slide ' . ($i + 1) . ' of ' . $n . ': ' . esc_attr(wp_strip_all_tags(str_replace("\n", ' ', $s['title']))) . '"' . (0 === $i ? ' aria-current="true" class="is-on"' : '') . '></button>';
        $o .= '</div><button class="ecn-hero-pause" type="button" aria-label="Pause slideshow" aria-pressed="false"><svg class="i-pause" viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" focusable="false"><rect x="7" y="6" width="3.2" height="12" rx="1"/><rect x="13.8" y="6" width="3.2" height="12" rx="1"/></svg><svg class="i-play" viewBox="0 0 24 24" fill="currentColor" aria-hidden="true" focusable="false"><path d="M8 6.5v11a1 1 0 0 0 1.5.86l9-5.5a1 1 0 0 0 0-1.72l-9-5.5A1 1 0 0 0 8 6.5Z"/></svg></button></div>';
    }
    $o .= '</section>';
    if ($n > 1) $o .= econur_hero_js();
    return $o;
});

function econur_hero_js() {
    return '<script>(function(){var s=document.currentScript&&document.currentScript.previousElementSibling;if(!s||!s.classList.contains("ecn-hero"))return;
var sl=[].slice.call(s.querySelectorAll(".ecn-hero-slide")),d=[].slice.call(s.querySelectorAll(".ecn-hero-dots button")),pb=s.querySelector(".ecn-hero-pause"),n=sl.length,cur=0,timer=null,hover=false,paused=false,DELAY=5500,
rm=window.matchMedia&&window.matchMedia("(prefers-reduced-motion: reduce)").matches;if(rm){paused=true;s.classList.add("is-paused");pb.setAttribute("aria-pressed","true");pb.setAttribute("aria-label","Play slideshow");}
function wake(el){[].forEach.call(el.querySelectorAll("img[loading=lazy]"),function(i){i.loading="eager";});}
function go(i){i=(i+n)%n;if(i===cur)return;var a=sl[cur],b=sl[i];wake(b);a.classList.remove("is-on");a.setAttribute("aria-hidden","true");a.inert=true;b.classList.add("is-on");b.removeAttribute("aria-hidden");b.inert=false;
d.forEach(function(x,k){x.classList.toggle("is-on",k===i);if(k===i){x.setAttribute("aria-current","true");}else{x.removeAttribute("aria-current");}});cur=i;wake(sl[(i+1)%n]);}
function stop(){if(timer){clearTimeout(timer);timer=null;}}function run(){stop();if(paused||hover||document.hidden)return;timer=setTimeout(function(){go(cur+1);run();},DELAY);}
d.forEach(function(b,k){b.addEventListener("click",function(){go(k);run();});});
s.querySelector(".ecn-hero-prev").addEventListener("click",function(){go(cur-1);run();});s.querySelector(".ecn-hero-next").addEventListener("click",function(){go(cur+1);run();});
pb.addEventListener("click",function(){paused=!paused;s.classList.toggle("is-paused",paused);pb.setAttribute("aria-pressed",paused?"true":"false");pb.setAttribute("aria-label",paused?"Play slideshow":"Pause slideshow");run();});
s.addEventListener("mouseenter",function(){hover=true;stop();});s.addEventListener("mouseleave",function(){hover=false;run();});
s.addEventListener("focusin",function(){hover=true;stop();});s.addEventListener("focusout",function(e){if(!s.contains(e.relatedTarget)){hover=false;run();}});
s.addEventListener("keydown",function(e){if(e.key==="ArrowRight"){go(cur+1);run();}else if(e.key==="ArrowLeft"){go(cur-1);run();}});
var x0=null,y0=null;s.addEventListener("touchstart",function(e){var t=e.touches[0];x0=t.clientX;y0=t.clientY;hover=true;stop();},{passive:true});
s.addEventListener("touchend",function(e){hover=false;if(x0!==null){var t=e.changedTouches[0],dx=t.clientX-x0,dy=t.clientY-y0;if(Math.abs(dx)>40&&Math.abs(dx)>Math.abs(dy)*1.2){go(dx<0?cur+1:cur-1);}}x0=null;run();},{passive:true});
document.addEventListener("visibilitychange",run);
if("IntersectionObserver" in window){new IntersectionObserver(function(es){es.forEach(function(e){if(e.isIntersecting){run();}else{stop();}});},{threshold:.25}).observe(s);}else{run();}
window.addEventListener("load",function(){wake(sl[1%n]);});})();</script>';
}

add_action('admin_post_econur_hero_save', function () {
    if (!current_user_can('manage_options')) wp_die('Not allowed');
    check_admin_referer('econur_hero_save');
    $rows = isset($_POST['s']) && is_array($_POST['s']) ? wp_unslash($_POST['s']) : array();
    $slides = array();
    foreach ($rows as $r) {
        $d = absint($r['desktop'] ?? 0);
        if (!$d) continue;
        $slides[] = array(
            'order' => intval($r['order'] ?? 0), 'on' => empty($r['on']) ? 0 : 1, 'desktop' => $d, 'mobile' => absint($r['mobile'] ?? 0),
            'eyebrow' => sanitize_text_field($r['eyebrow'] ?? ''), 'title' => sanitize_textarea_field($r['title'] ?? ''),
            'text' => sanitize_textarea_field($r['text'] ?? ''), 'cta' => sanitize_text_field($r['cta'] ?? ''), 'url' => esc_url_raw(trim($r['url'] ?? '')),
        );
    }
    usort($slides, function ($a, $b) { return $a['order'] - $b['order']; });
    foreach ($slides as &$x) unset($x['order']);
    unset($x);
    update_option('econur_hero_slides', $slides, false);
    do_action('litespeed_purge_all');
    wp_safe_redirect(add_query_arg(array('page' => 'econur-hero-slides', 'saved' => 1), admin_url('themes.php')));
    exit;
});
