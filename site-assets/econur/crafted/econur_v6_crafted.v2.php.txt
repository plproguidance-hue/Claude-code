        array('pin', 'Made in Bangladesh', 'Locally crafted with thoughtful care.'),

    $rows = '';

        . '<a class="v8-craft-cta" href="' . esc_url(wc_get_page_permalink('shop')) . '"><span>Shop all bars</span>' . econur_v6_icon('arrow') . '</a></div></div>';
}
